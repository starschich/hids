\begin{frame}{AC power transmission --- the simplified relation}
  \begin{block}{Principle}
    {Active power flows on the \emph{angle} between two voltages\\
    Reactive power on the \emph{difference in their magnitude}.}
  \end{block}

  \vspace{1em}
  \[
    \begin{aligned}
      P   &= \frac{U_S\cdot U_R}{X}\cdot\sin\delta \\[2ex]
      Q_R &= \frac{U_S\cdot U_R\cdot\cos\delta - U_R^2}{X}
    \end{aligned}
  \]
\end{frame}
