      {\small
      \begin{tabularx}{\linewidth}{@{}l@{\hspace{0.6em}}X@{}}
        $P_S$ & is the export active power of the VSC converter station at PCC\\
        $Q_S$ & is the export reactive power of the VSC converter station at PCC\\
        $U_S$ & is the RMS value of the fundamental frequency component of the line-to-line AC voltage of the VSC converter station at PCC\\
        $U_E$ & is the equivalent AC voltage of the AC grid, corresponding to a SCL at PCC\\
        $R_S$ & is the equivalent resistance of the SCL impedance at PCC\\
        $X_S$ & is the equivalent reactance of the SCL impedance at PCC\\
        $\delta$ & is the phase angle difference between AC voltages $U_S$ and $U_E$\\
      \end{tabularx}}\\[0.3em]
      {\color{atGray}\footnotesize IEC TR 63363-1:2022}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Power exchange in the AC grid}
  \centering
  \topoimg[width=\linewidth,height=0.4\textheight,keepaspectratio]{\acfig{image4.pdf}}\\[0.3em]
  \vspace{1em}     
  $P_S = \frac{U_S}{R_{S}^{2}+X_{S}^{2}} \cdot \left[ \left ( U_S - U_E \cdot \cos \delta \right )  \cdot R_S + U_E \cdot X_S \cdot \sin \delta \right]$\\
  \vspace{1em}
  $Q_S = \frac{U_S}{R_{S}^{2}+X_{S}^{2}} \cdot \left[ \left ( U_S - U_E \cdot \cos \delta \right )  \cdot X_S - U_E \cdot R_S \cdot \sin \delta \right]$\\

\end{frame}

The calculation of the 2-port units is based on two paratemers which are defined by the physical set-up of the transmission:
\begin{itemize}
  \item Propagation constant
  \item Wave impedance
\end{itemize}
\end{frame}

\begin{frame}{2-port units --- line constants}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.40\textwidth}\centering
      \input{\acfig{image18.tex}}
    \end{column}
    \begin{column}{0.57\textwidth}
      Propagation constant\\
      $\begin{aligned}[t]
        \underline{\gamma}&=\alpha+j\cdot\beta =\sqrt{\underline{Z}'\cdot\underline{Y}'}\\[0.4em]
                          &=\sqrt{\left(R'\left(\omega \right)+ j\cdot\omega\cdot L'\left(\omega \right)\right)\cdot\left(G'+j \cdot \omega \cdot C'\right)}
       \end{aligned}$\\[2em]
      Wave impedance\\
      $\begin{aligned}[t]
         \underline{Z}_w &= \sqrt{\frac{\underline{Z}'}{\underline{Y}'}} = \sqrt{\frac{R'\left(\omega \right)+ j\cdot\omega\cdot L'\left(\omega \right)}{G' + j\cdot\omega\cdot C'}}
       \end{aligned}$\\
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2-port units --- line equations}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.40\textwidth}\centering
      \input{\acfig{image18.tex}}
    \end{column}
    \begin{column}{0.57\textwidth}
      Voltage along line, at distance $x$ from the sending end\\
      $\begin{aligned}[t]
        \underline{U} &= \underline{U}_2\cosh\underline{\gamma}\left(l-x\right)
                       + \underline{Z}_w\,\underline{I}_2\sinh\underline{\gamma}\left(l-x\right)\\[0.4em]
        \underline{I} &= \frac{\underline{U}_2}{\underline{Z}_w}\sinh\underline{\gamma}\left(l-x\right)
                       + \underline{I}_2\cosh\underline{\gamma}\left(l-x\right)
       \end{aligned}$\\[2em]
      2-port parameters, at the sending end $\left(x=0\right)$\\
      $\begin{aligned}[t]
        \underline{U}_1 &= \underline{U}_2\cosh\underline{\gamma}l
                         + \underline{Z}_w\,\underline{I}_2\sinh\underline{\gamma}l\\[0.4em]
        \underline{I}_1 &= \frac{\underline{U}_2}{\underline{Z}_w}\sinh\underline{\gamma}l
                         + \underline{I}_2\cosh\underline{\gamma}l
       \end{aligned}$\\
    \end{column}
  \end{columns}
\end{frame}
