% !TEX root = ../../presentation.tex
% =============================================================
%  TikZ — nominal pi-equivalent of a transmission line
%  Series Z = (R' + j w L') l between the rails, half the shunt
%  admittance Y = (G' + j w C') l at each end.  Terminal quantities
%  U_S/I_S and U_R/I_R match fig/two-port-terminals.tex.
%  The voltage arrows are inset so they do not touch the terminals.
%  Self-contained snippet -- \input directly inside a frame.
%  Drawn at natural size (x = y = 0.52 cm per unit, about 5 cm wide)
%  so the labels stay readable; do not wrap it in \resizebox.
% =============================================================
\begin{tikzpicture}[
    x=0.52cm, y=0.52cm,
    line width=0.7pt,
    every node/.style={inner sep=1.5pt, font=\small},
    rail/.style={atInk, line width=0.7pt},
    box/.style={draw=atInk, fill=atBox, line width=0.7pt,
                minimum width=1.0cm, minimum height=0.42cm},
    vbox/.style={draw=atInk, fill=atBox, line width=0.7pt,
                 minimum width=0.42cm, minimum height=0.85cm},
    term/.style={circle, draw=atInk, fill=white, line width=0.7pt,
                 minimum size=3.6pt, inner sep=0pt},
    flow/.style={-{Triangle[length=2.2mm,width=1.9mm]}, atInk,
                 line width=0.7pt},
  ]

  % ---- rails ------------------------------------------------
  \draw[rail] (0,2.6) -- (2.0,2.6);
  \draw[rail] (4.0,2.6) -- (6,2.6);
  \draw[rail] (0,0) -- (6,0);

  % ---- series branch ----------------------------------------
  \node[box] at (3,2.6) {$\underline{Z}$};

  % ---- shunt branches ---------------------------------------
  \draw[rail] (1.3,2.6) -- (1.3,2.0);
  \node[vbox] at (1.3,1.3) {$\frac{\underline{Y}}{2}$};
  \draw[rail] (1.3,0.6) -- (1.3,0);
  \draw[rail] (4.7,2.6) -- (4.7,2.0);
  \node[vbox] at (4.7,1.3) {$\frac{\underline{Y}}{2}$};
  \draw[rail] (4.7,0.6) -- (4.7,0);

  % ---- terminals --------------------------------------------
  \node[term] at (0,2.6) {};  \node[term] at (0,0) {};
  \node[term] at (6,2.6) {};  \node[term] at (6,0) {};

  % ---- terminal currents ------------------------------------
  \draw[flow] (-1.1,2.6) -- (-0.45,2.6);
  \node[left] at (-1.25,2.6) {$\underline{I}_S$};
  \draw[flow] (6.45,2.6) -- (7.1,2.6);
  \node[right] at (7.25,2.6) {$\underline{I}_R$};

  % ---- terminal voltages ------------------------------------
  \draw[flow] (0,2.15) -- (0,0.45);
  \node[left=2pt] at (0,1.3) {$\underline{U}_S$};
  \draw[flow] (6,2.15) -- (6,0.45);
  \node[right=2pt] at (6,1.3) {$\underline{U}_R$};

\end{tikzpicture}%
